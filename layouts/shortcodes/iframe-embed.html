{{/* Embed a self-contained interactive HTML report (e.g. a Listening Timeline plot) in a post. The HTML file lives under /static so Hugo copies it verbatim instead of treating it as a content page. Usage: {{< iframe-embed src="embeds/listening/map.html" plot="map" title="Where my listening comes from" >}} Parameters: src path under /static (required) plot value for the report's #plot= hash (optional) title accessible title + caption (optional; defaults to heading) heading replaces the plot's own heading text; heading="" hides it (use when the post has its own section heading for the plot) description replaces the plot's own sub-heading/description text; description="" hides it (optional; omit to keep the original) height starting height in px before the report reports its own (default 640) wide "true" breaks out of the text column, up to 1100px (default false: plots sit in the normal post column like any other figure) caption "true" shows the title as a caption under the plot (default false) The iframe follows the site's light/dark toggle by passing theme= in the hash, and resizes itself from the report's postMessage height. */}} {{- $src := .Get "src" -}} {{- $plot := .Get "plot" -}} {{- $heading := .Get "heading" -}} {{- $hasHeading := isset .Params "heading" -}} {{- $title := .Get "title" | default $heading | default "Interactive figure" -}} {{- $hasDesc := isset .Params "description" -}} {{- $height := .Get "height" | default "640" -}} {{- $wide := eq (printf "%v" (.Get "wide")) "true" -}} {{- $caption := eq (printf "%v" (.Get "caption")) "true" -}} {{- $url := relURL $src -}} {{- $hash := cond (ne $plot "") (printf "plot=%s&embed" $plot) "embed" -}} {{- if not (.Page.Store.Get "iframeEmbedInit") }} {{- .Page.Store.Set "iframeEmbedInit" true }} {{- end }}
